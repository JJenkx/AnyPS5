if(BUILD_TESTING)
    add_test_executable(regmgr_integer_tests tests/Integer.cpp)
    target_include_directories(regmgr_integer_tests PRIVATE ${LIBS_INCLUDE_DIR})
    target_link_libraries(regmgr_integer_tests PRIVATE libSceRegMgr)
    configure_windows_unwind(regmgr_integer_tests)
    foreach(registryCase IN ITEMS empty populated invalid duplicate)
        add_test(NAME regmgr_integer_${registryCase} COMMAND regmgr_integer_tests ${registryCase})
        if(registryCase STREQUAL "empty")
            set_tests_properties(regmgr_integer_${registryCase} PROPERTIES ENVIRONMENT_MODIFICATION "APS5_REGMGR_INTS=unset:")
        else()
            set_tests_properties(regmgr_integer_${registryCase} PROPERTIES ENVIRONMENT "APS5_REGMGR_INTS=${CMAKE_CURRENT_SOURCE_DIR}/tests/${registryCase}.txt")
        endif()
        if(WIN32)
            set_property(TEST regmgr_integer_${registryCase} APPEND PROPERTY ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceRegMgr>")
        endif()
    endforeach()
endif()
